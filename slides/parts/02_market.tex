\begin{frame}{VaR, ES and the FRTB rules}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \[ \VaR_\alpha(L) = \inf\{x : P(L \le x) \ge \alpha\}, \]
      \[ \ES_\alpha(L) = \tfrac{1}{1-\alpha}\textstyle\int_\alpha^1 \VaR_u(L)\,\dd u, \]
      with $L$ the loss (currency) and $\alpha$ the confidence level.
      \begin{itemize}
        \item ES is coherent \citep{artzner1999coherent,acerbi2002coherence}; VaR is not sub-additive.
        \item VaR is elicitable; ES is not on its own \citep{gneiting2011making} but (VaR, ES) is jointly \citep{fissler2016higher}.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{FRTB internal models} \citep{bcbs2023mar33}:
      \begin{itemize}
        \item ES at 97.5\%, base horizon 10 days;
        \item liquidity horizons 10, 20, 40, 60, 120 days, combined by a cascade formula;
        \item calibrated to stress through a reduced factor set explaining $\ge 75\%$ of ES;
        \item multiplier 1.5 plus a backtesting add-on up to 0.5.
      \end{itemize}
      In the EU these rules apply from 1 January 2027.
    \end{column}
  \end{columns}
\end{frame}
